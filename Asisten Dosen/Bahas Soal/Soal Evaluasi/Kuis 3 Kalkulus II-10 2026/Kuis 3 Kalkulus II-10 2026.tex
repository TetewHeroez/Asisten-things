\begin{enumerate}
  \item \textbf{Fungsi Logaritma Natural} \hfill \textcolor{red}{\textbf{[Total: 25 poin]}}
        \begin{enumerate}
          \item[(a)] Tentukan nilai $a$ dan $b$:
                \begin{align*}
                  f(a) = 1 & \implies \ln(2a) + 1 = 1 \implies \ln(2a) = 0 \implies 2a = e^0 = 1 \implies a = \frac{1}{2} &  & \text{\textcolor{red}{\textbf{[5 poin]}}} \\
                  f(b) = 2 & \implies \ln(2b) + 1 = 2 \implies \ln(2b) = 1 \implies 2b = e^1 = e \implies b = \frac{e}{2} &  & \text{\textcolor{red}{\textbf{[5 poin]}}}
                \end{align*}

          \item[(b)] Luas daerah yang dibatasi kurva:
                \begin{align*}
                  L & = \int_{1/2}^{e/2} (\ln(2x) + 1) \,dx &  & \text{\textcolor{red}{\textbf{[5 poin]}}}
                \end{align*}
                Gunakan integral parsial. Misal $u = \ln(2x) + 1 \implies du = \frac{1}{x}\,dx$, dan $dv = dx \implies v = x$.
                \begin{align*}
                  \int (\ln(2x) + 1) \,dx & = x(\ln(2x) + 1) - \int x \left(\frac{1}{x}\right) \,dx                                                \\
                                          & = x\ln(2x) + x - x                                                                                     \\
                                          & = x\ln(2x)                                              &  & \text{\textcolor{red}{\textbf{[5 poin]}}}
                \end{align*}
                Maka, batas integrasinya:
                \begin{align*}
                  L & = \left[ x\ln(2x) \right]_{1/2}^{e/2}                                                                                                                                       \\
                    & = \left( \frac{e}{2}\ln(e) \right) - \left( \frac{1}{2}\ln(1) \right) = \left( \frac{e}{2} \cdot 1 \right) - 0 = \frac{e}{2} &  & \text{\textcolor{red}{\textbf{[5 poin]}}}
                \end{align*}
        \end{enumerate}

        \textbf{Ilustrasi:}
        \begin{center}
          \begin{tikzpicture}
            \begin{axis}[
                axis lines=middle, xlabel={$x$}, ylabel={$y$},
                xmin=0, xmax=2, ymin=-1, ymax=3,
                width=8cm, height=6cm, samples=100
              ]
              \addplot[name path=A, blue, thick, domain=0.1:1.5] {ln(2*x) + 1};
              \addplot[name path=B, domain=0.5:1.359, draw=none] {0};
              \addplot[gray!30] fill between[of=A and B, soft clip={domain=0.5:1.359}];
              \addplot[dashed] coordinates {(0.5,0) (0.5,1)};
              \addplot[dashed] coordinates {(1.359,0) (1.359,2)};
              \node at (axis cs: 0.5, -0.4) {$1/2$};
              \node at (axis cs: 1.359, -0.4) {$e/2$};
            \end{axis}
          \end{tikzpicture}
        \end{center}

  \item \textbf{Volume Benda Putar (Sumbu-$x$)} \hfill \textcolor{red}{\textbf{[Total: 25 poin]}}
        \begin{enumerate}
          \item[(a)] Sketsa daerah: \hfill \textcolor{red}{\textbf{[10 poin]}} \\
                Pada interval $\left[0, \frac{\pi}{4}\right]$, nilai $\cos(x) \ge \sin(x)$.

                \begin{center}
                  \begin{tikzpicture}

                    \end{axis}
                  \end{tikzpicture}
                \end{center}

          \item[(b)] Volume benda putar mengelilingi sumbu-$x$ (Metode Cincin):
                Jari-jari luar $R(x) = \cos(x)$ dan jari-jari dalam $r(x) = \sin(x)$.
                \begin{align*}
                  V & = \pi \int_{0}^{\pi/4} \left( [R(x)]^2 - [r(x)]^2 \right) \,dx                                                  &  & \text{\textcolor{red}{\textbf{[5 poin]}}} \\
                    & = \pi \int_{0}^{\pi/4} \left( \cos^2(x) - \sin^2(x) \right) \,dx                                                                                               \\
                    & = \pi \int_{0}^{\pi/4} \cos(2x) \,dx                                                                            &  & \text{\textcolor{red}{\textbf{[5 poin]}}} \\
                    & = \pi \left[ \frac{1}{2}\sin(2x) \right]_{0}^{\pi/4}                                                                                                           \\
                    & = \frac{\pi}{2} \left( \sin\left(\frac{\pi}{2}\right) - \sin(0) \right) = \frac{\pi}{2} (1 - 0) = \frac{\pi}{2} &  & \text{\textcolor{red}{\textbf{[5 poin]}}}
                \end{align*}
        \end{enumerate}

  \item \textbf{Panjang Kurva} \hfill \textcolor{red}{\textbf{[Total: 25 poin]}} \\
        Diketahui $y = \frac{x^3}{6} + \frac{1}{2x}$. Turunannya adalah $y' = \frac{x^2}{2} - \frac{1}{2x^2}$. \hfill \textcolor{red}{\textbf{[5 poin]}} \\
        Maka, $1 + (y')^2 = 1 + \left( \frac{x^2}{2} - \frac{1}{2x^2} \right)^2 = 1 + \frac{x^4}{4} - \frac{1}{2} + \frac{1}{4x^4} = \frac{x^4}{4} + \frac{1}{2} + \frac{1}{4x^4} = \left( \frac{x^2}{2} + \frac{1}{2x^2} \right)^2$. \hfill \textcolor{red}{\textbf{[5 poin]}}
        \begin{align*}
          L & = \int_1^2 \sqrt{1 + (y')^2} \,dx = \int_1^2 \sqrt{\left( \frac{x^2}{2} + \frac{1}{2x^2} \right)^2} \,dx = \int_1^2 \left( \frac{x^2}{2} + \frac{1}{2x^2} \right) \,dx             &  & \text{\textcolor{red}{\textbf{[5 poin]}}} \\
            & = \left[ \frac{x^3}{6} - \frac{1}{2x} \right]_1^2 = \left( \frac{8}{6} - \frac{1}{4} \right) - \left( \frac{1}{6} - \frac{1}{2} \right)                                            &  & \text{\textcolor{red}{\textbf{[5 poin]}}} \\
            & = \left( \frac{4}{3} - \frac{1}{4} \right) - \left( \frac{1}{6} - \frac{3}{6} \right) = \frac{13}{12} - \left( -\frac{2}{6} \right) = \frac{13}{12} + \frac{4}{12} = \frac{17}{12} &  & \text{\textcolor{red}{\textbf{[5 poin]}}}
        \end{align*}

  \item \textbf{Luas Permukaan Benda Putar} \hfill \textcolor{red}{\textbf{[Total: 25 poin]}} \\
        Fungsi $y = \frac{x^3}{3}$. Turunannya merujuk ke $y' = x^2$. \hfill \textcolor{red}{\textbf{[5 poin]}} \\
        $1 + (y')^2 = 1 + x^4$.
        \begin{align*}
          S & = \int_0^1 2\pi y \sqrt{1 + (y')^2} \,dx = \int_0^1 2\pi \left( \frac{x^3}{3} \right) \sqrt{1 + x^4} \,dx = \frac{2\pi}{3} \int_0^1 x^3 \sqrt{1 + x^4} \,dx &  & \text{\textcolor{red}{\textbf{[5 poin]}}}
        \end{align*}
        Gunakan substitusi $u = 1 + x^4$, sehingga $du = 4x^3 \,dx \implies x^3 \,dx = \frac{du}{4}$. \hfill \textcolor{red}{\textbf{[5 poin]}} \\
        Batas atas: $x = 1 \implies u = 2$. Batas bawah: $x = 0 \implies u = 1$.
        \begin{align*}
          S & = \frac{2\pi}{3} \int_1^2 \sqrt{u} \frac{du}{4} = \frac{\pi}{6} \int_1^2 u^{1/2} \,du                                                   &  & \text{\textcolor{red}{\textbf{[5 poin]}}} \\
            & = \frac{\pi}{6} \left[ \frac{2}{3} u^{3/2} \right]_1^2 = \frac{\pi}{9} \left( 2^{3/2} - 1^{3/2} \right) = \frac{\pi}{9} (2\sqrt{2} - 1) &  & \text{\textcolor{red}{\textbf{[5 poin]}}}
        \end{align*}

              \draw[->, thick, dashed] (axis cs: 0.5, 0.1) arc (90:270:0.05 and 0.1) node[pos=0, right] {putar sumbu-x};
            \end{axis}
          \end{tikzpicture}
        \end{center}
\end{enumerate}
