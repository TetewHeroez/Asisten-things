\begin{enumerate}
  \item \textbf{Fungsi Logaritma Natural}
        \begin{enumerate}
          \item[(a)] Tentukan nilai $a$ dan $b$:
                \begin{align*}
                  f(a) = 1 & \implies \ln(2a) + 1 = 1 \implies \ln(2a) = 0 \implies 2a = e^0 = 1 \implies a = \frac{1}{2} \\
                  f(b) = 2 & \implies \ln(2b) + 1 = 2 \implies \ln(2b) = 1 \implies 2b = e^1 = e \implies b = \frac{e}{2}
                \end{align*}

          \item[(b)] Luas daerah yang dibatasi kurva:
                \begin{align*}
                  L & = \int_{1/2}^{e/2} (\ln(2x) + 1) \,dx
                \end{align*}
                Gunakan integral parsial. Misal $u = \ln(2x) + 1 \implies du = \frac{1}{x}\,dx$, dan $dv = dx \implies v = x$.
                \begin{align*}
                  \int (\ln(2x) + 1) \,dx & = x(\ln(2x) + 1) - \int x \left(\frac{1}{x}\right) \,dx \\
                                          & = x\ln(2x) + x - x \\
                                          & = x\ln(2x)
                \end{align*}
                Maka, batas integrasinya:
                \begin{align*}
                  L & = \left[ x\ln(2x) \right]_{1/2}^{e/2} \\
                    & = \left( \frac{e}{2}\ln(e) \right) - \left( \frac{1}{2}\ln(1) \right) = \left( \frac{e}{2} \cdot 1 \right) - 0 = \frac{e}{2}
                \end{align*}
        \end{enumerate}

        \textbf{Ilustrasi:}
        \begin{center}
          \begin{tikzpicture}
            \begin{axis}[
                axis lines=middle, xlabel={$x$}, ylabel={$y$},
                xmin=0, xmax=2, ymin=-1, ymax=3,
                width=8cm, height=6cm, samples=100
              ]
              \addplot[name path=A, blue, thick, domain=0.1:1.5] {ln(2*x) + 1};
              \addplot[name path=B, domain=0.5:1.359, draw=none] {0};
              \addplot[gray!30] fill between[of=A and B, soft clip={domain=0.5:1.359}];
              \addplot[dashed] coordinates {(0.5,0) (0.5,1)};
              \addplot[dashed] coordinates {(1.359,0) (1.359,2)};
              \node at (axis cs: 0.5, -0.4) {$1/2$};
              \node at (axis cs: 1.359, -0.4) {$e/2$};
            \end{axis}
          \end{tikzpicture}
        \end{center}

  \item \textbf{Volume Benda Putar (Sumbu-$x$)}
        \begin{enumerate}
          \item[(a)] Sketsa daerah:
                Pada interval $\left[0, \frac{\pi}{4}\right]$, nilai $\cos(x) \ge \sin(x)$.
                
                \begin{center}
                  \begin{tikzpicture}
                    \begin{axis}[
                        axis lines=middle, xlabel={$x$}, ylabel={$y$},
                        xmin=-0.2, xmax=1.2, ymin=-0.2, ymax=1.2,
                        xtick={0, 0.785}, xticklabels={0, $\frac{\pi}{4}$},
                        width=8cm, height=6cm, samples=100
                      ]
                      \addplot[name path=cos, blue, thick, domain=0:0.785] {cos(deg(x))};
                      \addplot[name path=sin, red, thick, domain=0:0.785] {sin(deg(x))};
                      \addplot[gray!30] fill between[of=cos and sin];
                      \draw[->, thick, dashed] (axis cs: 0.4, 0.4) arc (90:270:0.2 and 0.4) node[midway, right] {putar sumbu-x};
                    \end{axis}
                  \end{tikzpicture}
                \end{center}

          \item[(b)] Volume benda putar mengelilingi sumbu-$x$ (Metode Cincin):
                Jari-jari luar $R(x) = \cos(x)$ dan jari-jari dalam $r(x) = \sin(x)$.
                \begin{align*}
                  V & = \pi \int_{0}^{\pi/4} \left( [R(x)]^2 - [r(x)]^2 \right) \,dx \\
                    & = \pi \int_{0}^{\pi/4} \left( \cos^2(x) - \sin^2(x) \right) \,dx \\
                    & = \pi \int_{0}^{\pi/4} \cos(2x) \,dx \\
                    & = \pi \left[ \frac{1}{2}\sin(2x) \right]_{0}^{\pi/4} \\
                    & = \frac{\pi}{2} \left( \sin\left(\frac{\pi}{2}\right) - \sin(0) \right) = \frac{\pi}{2} (1 - 0) = \frac{\pi}{2}
                \end{align*}
        \end{enumerate}

  \item \textbf{Panjang Kurva} \\
        Diketahui $y = \frac{2}{3}x^{3/2}$. Turunannya adalah $y' = x^{1/2}$. \\
        Maka, $1 + (y')^2 = 1 + x$.
        \begin{align*}
          L & = \int_0^3 \sqrt{1 + (y')^2} \,dx = \int_0^3 \sqrt{1 + x} \,dx                                       \\
            & = \left[ \frac{2}{3}(1 + x)^{3/2} \right]_0^3 = \frac{2}{3} \left( (1+3)^{3/2} - (1+0)^{3/2} \right) \\
            & = \frac{2}{3} (4^{3/2} - 1) = \frac{2}{3} (8 - 1) = \frac{14}{3}
        \end{align*}

        \textbf{Ilustrasi:}
        \begin{center}
          \begin{tikzpicture}
            \begin{axis}[
                axis lines=middle, xlabel={$x$}, ylabel={$y$},
                xmin=-0.2, xmax=3.5, ymin=-0.2, ymax=4.5,
                width=8cm, height=6cm, samples=100
              ]
              \addplot[blue, thick, domain=0:3] {2/3 * x^(1.5)};
              \draw[dashed] (axis cs: 3,0) -- (axis cs: 3, 3.464);
              \node at (axis cs: 1.5, 1.8) [anchor=east] {$y = \frac{2}{3}x^{3/2}$};
            \end{axis}
          \end{tikzpicture}
        \end{center}

  \item \textbf{Luas Permukaan Benda Putar} \\
        Fungsi $y = \sqrt{x}$. Turunannya merujuk ke $y' = \frac{1}{2\sqrt{x}}$. \\
        $1 + (y')^2 = 1 + \frac{1}{4x} = \frac{4x + 1}{4x}$.
        \begin{align*}
          S & = \int_0^2 2\pi y \sqrt{1 + (y')^2} \,dx = \int_0^2 2\pi \sqrt{x} \sqrt{\frac{4x+1}{4x}} \,dx                                                                                     \\
            & = \int_0^2 2\pi \sqrt{x} \frac{\sqrt{4x+1}}{2\sqrt{x}} \,dx = \pi \int_0^2 \sqrt{4x+1} \,dx                                                                                       \\
            & = \pi \left[ \frac{1}{4} \cdot \frac{2}{3} (4x+1)^{3/2} \right]_0^2 = \frac{\pi}{6} \left( 9^{3/2} - 1^{3/2} \right) = \frac{\pi}{6} (27 - 1) = \frac{26\pi}{6} = \frac{13\pi}{3}
        \end{align*}

        \textbf{Ilustrasi:}
        \begin{center}
          \begin{tikzpicture}
            \begin{axis}[
                axis lines=middle, xlabel={$x$}, ylabel={$y$},
                xmin=-0.2, xmax=2.5, ymin=-1.8, ymax=1.8,

              ]
              \addplot[blue, thick, domain=0:2] {sqrt(x)};
              \addplot[blue!50, dashed, domain=0:2] {-sqrt(x)};
              \draw[dashed, gray] (axis cs: 2, 1.414) arc (90:-90:0.3 and 1.414);
              \draw[gray] (axis cs: 2, -1.414) arc (-90:-270:0.3 and 1.414);
              \draw[->, thick, dashed] (axis cs: 1, 0.5) arc (90:270:0.2 and 0.5) node[midway, right] {putar sumbu-x};
            \end{axis}
          \end{tikzpicture}
        \end{center}
\end{enumerate}
