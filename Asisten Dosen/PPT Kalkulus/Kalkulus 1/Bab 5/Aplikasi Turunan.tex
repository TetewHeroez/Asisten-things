\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Susun hubungan sebelum memasukkan angka}

\begin{block}{Laju-laju yang Berkaitan}
Besaran-besaran yang bergantung pada waktu dihubungkan oleh persamaan. Turunkan persamaan itu terhadap $t$ menggunakan aturan rantai untuk menghubungkan laju perubahannya.
\end{block}
\begin{enumerate}
\item Gambar situasi dan beri nama besaran serta laju yang diketahui.
\item Tulis persamaan yang berlaku selama proses berlangsung.
\item Turunkan terhadap waktu, kemudian substitusikan keadaan yang ditanyakan.
\item Periksa tanda dan satuan hasilnya.
\end{enumerate}
\onslide<2->{\begin{alertblock}{Mengapa angka dimasukkan belakangan?}
$r=10$ mungkin hanya berlaku pada satu saat. Menggantikan $r(t)$ dengan konstanta sebelum menurunkan akan menghilangkan perubahan yang dicari.
\end{alertblock}}
\end{frame}

\section{Interval Naik, Turun, dan Kecekungan Fungsi}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Naik dan turun adalah sifat pada interval}

% Definisi 5.2.1, hlm. 147.
\begin{definisi}
Untuk setiap $x_1<x_2$ dalam interval $I$, fungsi $f$ disebut naik jika $f(x_1)<f(x_2)$, turun jika $f(x_1)>f(x_2)$, dan konstan jika $f(x_1)=f(x_2)$ untuk semua pasangan titiknya.
\end{definisi}
\onslide<2->{\begin{block}{Baca grafik dari kiri ke kanan}
Yang dibandingkan adalah tinggi grafik, bukan apakah nilainya positif atau negatif. Fungsi $f(x)=x-10$ tetap naik ketika nilainya negatif.
\end{block}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Tanda turunan pertama menentukan kemonotonan}

% Teorema 5.2.1, hlm. 147.
\begin{teorema}
Jika $f$ kontinu pada $[a,b]$ dan dapat diturunkan pada $(a,b)$:
\begin{itemize}
\item $f'>0$ di seluruh $(a,b)$ menjamin $f$ naik pada $[a,b]$;
\item $f'<0$ menjamin $f$ turun;
\item $f'=0$ di seluruh interval menjamin $f$ konstan.
\end{itemize}
\end{teorema}
\onslide<2->{\begin{alertblock}{Satu titik datar belum berarti konstan}
Untuk $f(x)=x^3$, $f'(0)=0$, tetapi fungsi tetap naik pada $\mathbb R$. Syarat untuk konstan berlaku di seluruh interval.
\end{alertblock}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Contoh: tabel tanda sebelum menggambar}

% Modifikasi Contoh 5.2.2, hlm. 149.
\begin{contoh}Tentukan interval naik dan turun $f(x)=x^3-6x^2+5$.\end{contoh}
\onslide<2->{\[f'(x)=3x(x-4).\]
\begin{center}\begin{tabular}{c|ccccc}
$x$&$(-\infty,0)$&$0$&$(0,4)$&$4$&$(4,\infty)$\\\hline
$f'$&$+$&$0$&$-$&$0$&$+$\\
$f$&$\nearrow$&&$\searrow$&&$\nearrow$
\end{tabular}\end{center}
Karena kontinu, $f$ naik pada $(-\infty,0]$ dan $[4,\infty)$, serta turun pada $[0,4]$.
\begin{block}{Mengapa satu titik uji cukup?}
Faktor-faktor $3x(x-4)$ tidak berubah tanda di antara akar-akarnya.
\end{block}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Kecekungan mengukur perubahan kemiringan}

% Definisi 5.2.2, hlm. 150.
\begin{definisi}
Pada interval tempat $f$ dapat diturunkan, grafik cekung ke atas jika $f'$ naik, dan cekung ke bawah jika $f'$ turun.
\end{definisi}\begin{columns}[c]
\begin{column}{.48\textwidth}

\centering\begin{tikzpicture}[scale=.7]
\draw[->] (-1.8,0)--(1.8,0);
\draw[HIMAtua,very thick,domain=-1.5:1.5] plot (\x,{.6*\x*\x+.3});
\foreach \a in {-1,0,1}{\draw[red,domain=-.4:.4] plot ({\a+\x},{.6*\a*\a+.3+1.2*\a*\x});}
\node at (0,-.5) {kemiringan bertambah};\end{tikzpicture}
\end{column}
\begin{column}{.48\textwidth}

\onslide<2->{\begin{alertblock}{Naik bukan berarti cekung ke atas}
$\sqrt{x}$ naik, tetapi gradiennya mengecil. Grafiknya cekung ke bawah untuk $x>0$.
\end{alertblock}}
\end{column}
\end{columns}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Turunan kedua dan titik belok}

% Teorema 5.2.2 dan Definisi 5.2.3, hlm. 151.
\begin{teorema}
Jika $f''>0$ pada interval terbuka, grafik $f$ cekung ke atas. Jika $f''<0$, grafik cekung ke bawah.
\end{teorema}
\begin{definisi}
Jika $f$ kontinu di sekitar $c$ dan kecekungannya berubah saat melewati $c$, maka $(c,f(c))$ disebut titik belok.
\end{definisi}
\onslide<2->{\begin{alertblock}{Tidak cukup hanya $f''(c)=0$}
$f(x)=x^4$ memenuhi $f''(0)=0$, tetapi kecekungannya tidak berubah. Jadi $(0,0)$ bukan titik belok.
\end{alertblock}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Contoh: menggabungkan dua jenis informasi}
\begin{columns}[c]
\begin{column}{.48\textwidth}
\centering\begin{tikzpicture}\begin{axis}[babfive,domain=-2:2,xmin=-2.2,xmax=2.2,ymin=-3,ymax=3,xtick={-1,0,1},ytick={-2,2}]
\addplot[HIMAtua,very thick] {x^3-3*x};
\addplot[only marks,red] coordinates {(-1,2) (0,0) (1,-2)};
\end{axis}\end{tikzpicture}
\end{column}
\begin{column}{.48\textwidth}

% Adaptasi Contoh 5.2.3, hlm. 151--153.
\begin{contoh}$f(x)=x^3-3x$.\end{contoh}
\[f'=3(x-1)(x+1),\qquad f''=6x.\]
\onslide<2->{Naik untuk $x<-1$ atau $x>1$; turun untuk $-1<x<1$.
Cekung ke bawah untuk $x<0$, ke atas untuk $x>0$.
Titik belok $(0,0)$ terjadi ketika fungsi sedang turun.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Latihan: membaca grafik turunannya}

\begin{exampleblock}{Latihan}
\begin{enumerate}
\item Cari interval naik, turun, dan kecekungan $f(x)=x^3-3x^2+2$.
\item Apakah setiap titik dengan $f'(c)=0$ merupakan titik belok? Bandingkan $x^2$ dan $x^3$.
\item Jika grafik $f'$ berada di atas sumbu $x$ tetapi menurun, apa arti kedua informasi tersebut bagi $f$?
\end{enumerate}\end{exampleblock}
\onslide<2->{\textbf{Kunci membaca:} tinggi grafik $f'$ menentukan tanda $f'$, sedangkan kemiringannya menentukan tanda $f''$.}
\end{frame}

\section{Ekstrim Relatif, Uji Turunan Pertama dan Kedua}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Ekstrim relatif membandingkan lingkungan sekitar}

% Definisi 5.3.1, hlm. 157.
\begin{definisi}
$f$ mempunyai maksimum relatif di $c$ jika $f(c)\ge f(x)$ untuk semua $x$ dalam suatu interval terbuka yang memuat $c$. Minimum relatif didefinisikan dengan $f(c)\le f(x)$. Keduanya disebut ekstrim relatif.
\end{definisi}
\onslide<2->{\begin{block}{Puncak lokal belum tentu paling tinggi}
Pada $f(x)=x^3-3x$, titik $(-1,2)$ adalah maksimum relatif. Namun $f(3)=18$ lebih besar: fungsi tidak mempunyai maksimum di seluruh $\mathbb R$.
\end{block}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Titik kritis adalah kandidat}

% Teorema 5.3.1 dan Definisi 5.3.2, hlm. 157.
\begin{teorema}
Jika $f$ mempunyai ekstrim relatif di titik interior $c$, maka $f'(c)=0$ atau $f'(c)$ tidak ada.
\end{teorema}
\begin{definisi}
Titik kritis adalah $c$ dalam domain $f$ dengan $f'(c)=0$ atau $f'(c)$ tidak ada. Titik dengan $f'(c)=0$ disebut titik stasioner.
\end{definisi}
\onslide<2->{\textbf{Bandingkan di $0$:} $x^2$ minimum stasioner; $|x|$ minimum tak terturunkan; $x^3$ stasioner tetapi bukan ekstrim.}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Mengapa ekstrim yang mulus harus datar?}

\begin{block}{Gagasan Bukti untuk Maksimum Relatif}
Jika $f(c+h)\le f(c)$ di dekat $c$, maka
\[\frac{f(c+h)-f(c)}h\le0\ (h>0),\qquad
\frac{f(c+h)-f(c)}h\ge0\ (h<0).\]
Jika turunan ada, kedua limit sama. Satu tidak positif dan satu tidak negatif, sehingga $f'(c)=0$.
\end{block}
\onslide<2->{\begin{alertblock}{Perhatikan syaratnya}
Bukti membutuhkan turunan yang ada dan titik interior. Titik sudut dan titik ujung harus diperiksa tersendiri.
\end{alertblock}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Uji turunan pertama: perubahan tanda}

% Teorema 5.3.2, hlm. 158.
\begin{teorema}[Uji Turunan Pertama]
Misalkan $f$ kontinu di $c$ dan dapat diturunkan pada kedua sisi dekat $c$.
\begin{center}\begin{tabular}{c|c|l}
$f'$ di kiri&$f'$ di kanan&Kesimpulan di $c$\\\hline
$+$&$-$&Maksimum relatif\\
$-$&$+$&Minimum relatif\\
$+$&$+$&Bukan ekstrim relatif\\
$-$&$-$&Bukan ekstrim relatif
\end{tabular}\end{center}\end{teorema}
\onslide<2->{\textbf{Maknanya:} naik lalu turun menghasilkan puncak; turun lalu naik menghasilkan lembah. Uji ini juga dapat digunakan pada titik sudut.}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Uji turunan kedua dan batasnya}

% Teorema 5.3.3, hlm. 159.
\begin{teorema}[Uji Turunan Kedua]
Misalkan $f$ dapat diturunkan dua kali di sekitar titik stasioner $c$. Jika $f''(c)<0$, maka $f(c)$ maksimum relatif; jika $f''(c)>0$, maka $f(c)$ minimum relatif.
\end{teorema}
\onslide<2->{\begin{alertblock}{Jika $f''(c)=0$, uji belum memberi jawaban}
Di $0$, ketiganya mempunyai turunan kedua nol:
\[x^4\text{ (minimum)},\quad -x^4\text{ (maksimum)},\quad x^3\text{ (bukan ekstrim)}.\]
Kembali ke perubahan tanda turunan pertama.
\end{alertblock}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Contoh: dua lembah dan satu puncak}
\begin{columns}[c]
\begin{column}{.48\textwidth}
\centering\begin{tikzpicture}\begin{axis}[babfive,domain=-3:3,xmin=-3,xmax=3,ymin=-19,ymax=6,xtick={-2,0,2},ytick={-16,0}]
\addplot[HIMAtua,very thick] {x^4-8*x^2};
\addplot[only marks,red] coordinates {(-2,-16) (0,0) (2,-16)};
\end{axis}\end{tikzpicture}
\end{column}
\begin{column}{.48\textwidth}

% Modifikasi Contoh 5.3.5, hlm. 160--161.
\begin{contoh}$f(x)=x^4-8x^2$.\end{contoh}
\[f'=4x(x-2)(x+2),\quad f''=12x^2-16.\]
\onslide<2->{Titik stasioner: $-2,0,2$.
$f''(0)<0$: maksimum relatif $(0,0)$.
$f''(\pm2)>0$: minimum relatif $(\pm2,-16)$.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Contoh: ekstrim tanpa turunan}

% Modifikasi Contoh 5.3.3, hlm. 160.
\begin{contoh}Tentukan ekstrim relatif $f(x)=3x^{5/3}-30x^{2/3}$.\end{contoh}
\onslide<2->{Untuk $x\ne0$, $f'(x)=5x^{-1/3}(x-4)$.
Tandanya positif pada $(-\infty,0)$, negatif pada $(0,4)$, dan positif pada $(4,\infty)$.
\begin{block}{Kesimpulan}
Maksimum relatif $(0,0)$ meskipun $f'(0)$ tidak ada.
Minimum relatif $(4,-18\cdot4^{2/3})$.
\end{block}
Titik $0$ tetap termasuk kandidat karena $f(0)$ terdefinisi.}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Latihan: kandidat belum tentu ekstrim}

\begin{exampleblock}{Latihan}\begin{enumerate}
\item Klasifikasikan titik kritis $f(x)=(x-2)^3$.
\item Tentukan ekstrim relatif $f(x)=x^4-4x^2+1$ dengan kedua uji.
\item Tentukan ekstrim relatif $f(x)=|x^2-1|$. Periksa juga titik yang tidak dapat diturunkan.
\end{enumerate}\end{exampleblock}
\onslide<2->{\textbf{Pertanyaan tambahan:} apakah bilangan yang membuat penyebut $f$ nol merupakan titik kritis? Periksa terlebih dahulu domain fungsi.}
\end{frame}

\section{Grafik Polinomial dan Fungsi Rasional}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Sketsa grafik adalah rangkuman analisis}

% Hlm. 163--164.
\begin{block}{Urutan Pemeriksaan}
\begin{enumerate}
\item Domain, simetri, dan perpotongan dengan sumbu koordinat.
\item Limit di batas domain dan perilaku ketika $x\to\pm\infty$.
\item Titik kritis dan tanda $f'$ untuk naik, turun, dan ekstrim.
\item Tanda $f''$ untuk kecekungan dan titik belok.
\item Gabungkan semuanya dalam sketsa; periksa beberapa titik.
\end{enumerate}\end{block}
\onslide<2->{\begin{alertblock}{Grafik komputer juga perlu diperiksa}
Jendela gambar yang terlalu sempit dapat menyembunyikan titik belok atau ekstrim. Analisis fungsi menjadi dasar untuk memilih skala gambar.
\end{alertblock}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Contoh: sketsa polinomial}
\begin{columns}[c]
\begin{column}{.48\textwidth}
\centering\begin{tikzpicture}\begin{axis}[babfive,domain=-2.2:2.2,xmin=-2.2,xmax=2.2,ymin=-4,ymax=5,xtick={-1,0,1},ytick={-1,1,3}]
\addplot[HIMAtua,very thick] {x^3-3*x+1};
\addplot[only marks,red] coordinates {(-1,3) (0,1) (1,-1)};
\end{axis}\end{tikzpicture}
\end{column}
\begin{column}{.48\textwidth}

% Modifikasi Contoh 5.4.1, hlm. 164--165.

\framesubtitle{Asimtot tegak dan datar}

% Definisi 5.4.1, hlm. 165.
\begin{definisi}

\section{Nilai Maksimum dan Minimum Suatu Fungsi}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Ekstrim absolut bergantung pada domain}

% Definisi 5.5.1, hlm. 175.
\begin{definisi}
Pada himpunan $I$, $f(c)$ adalah maksimum absolut jika $f(c)\ge f(x)$ untuk semua $x\in I$, dan minimum absolut jika $f(c)\le f(x)$ untuk semua $x\in I$.
\end{definisi}
\onslide<2->{\begin{contoh}
Untuk $f(x)=x^2$ pada $[-1,2]$, minimum adalah $0$ dan maksimum $4$. Pada $\mathbb R$, minimum tetap $0$, tetapi maksimum tidak ada.
\end{contoh}
\textbf{Tiga pertanyaan berbeda:} apakah ekstrim ada, berapa nilainya, dan di titik mana nilai itu dicapai?}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Jaminan adanya ekstrim}

% Teorema 5.5.1, hlm. 175.
\begin{teorema}[Teorema Nilai Ekstrim]
Jika $f$ kontinu pada interval tertutup dan terbatas $[a,b]$, maka $f$ mempunyai maksimum dan minimum pada interval tersebut.
\end{teorema}
\onslide<2->{\begin{columns}[t]
\begin{column}{.48\textwidth}\begin{block}{Jika ujung tidak termasuk}
$f(x)=x$ pada $(0,1)$ tidak mencapai $0$ atau $1$: maksimum dan minimum tidak ada.
\end{block}\end{column}
\begin{column}{.48\textwidth}\begin{block}{Jika tidak kontinu}
$f(0)=0$, $f(x)=1/x$ untuk $0<x\le1$ tidak memiliki maksimum pada $[0,1]$.
\end{block}\end{column}\end{columns}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Di mana ekstrim dapat terjadi?}

% Teorema 5.5.2, hlm. 175--176.
\begin{teorema}
Jika maksimum atau minimum dicapai di bagian dalam suatu interval, maka titik tempatnya terjadi adalah titik kritis.
\end{teorema}
\begin{block}{Metode Interval Tertutup}
Untuk $f$ kontinu pada $[a,b]$: cari titik kritis dalam $(a,b)$, hitung nilai fungsi di sana dan di kedua ujung, lalu bandingkan semuanya.
\end{block}
\onslide<2->{\begin{alertblock}{Uji lokal saja tidak cukup}
Maksimum relatif dapat kalah tinggi dari nilai di ujung. Yang dibandingkan adalah nilai $f$, bukan $f'$.
\end{alertblock}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Contoh: tabel kandidat termasuk titik ujung}

% Modifikasi Contoh 5.5.7, hlm. 179.
\begin{contoh}Cari ekstrim absolut $f(x)=x^3-3x^2+1$ pada $[-1,3]$.\end{contoh}
\onslide<2->{\[f'(x)=3x(x-2)=0\Rightarrow x=0,2.\]
\begin{center}\begin{tabular}{c|rrrr}
$x$&$-1$&$0$&$2$&$3$\\\hline
$f(x)$&$-3$&$1$&$-3$&$1$
\end{tabular}\end{center}
Minimum $-3$ di $x=-1,2$; maksimum $1$ di $x=0,3$.
\begin{block}{Dapat dicapai lebih dari sekali}
Titik ujung dan titik interior dapat memberikan nilai ekstrim yang sama.
\end{block}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Contoh: jangan lupakan titik tak terturunkan}

% Modifikasi Contoh 5.5.8, hlm. 180.
\begin{contoh}Cari ekstrim $f(x)=3x^{4/3}-4x^{1/3}$ pada $[-1,1]$.\end{contoh}
\onslide<2->{\[f'(x)=\frac{4(3x-1)}{3x^{2/3}},\quad x\ne0.\]
Kandidat: ujung $-1,1$, titik stasioner $1/3$, dan titik tak terturunkan $0$.
\[f(-1)=7,\quad f(0)=0,\quad f(1/3)=-\sqrt[3]9,\quad f(1)=-1.\]
Maksimum $7$ di $-1$; minimum $-\sqrt[3]9$ di $1/3$.
\begin{alertblock}{Periksa domain}
Akar kubik terdefinisi untuk bilangan negatif. Titik $0$ juga termasuk domain fungsi meskipun turunannya tidak ada.
\end{alertblock}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Satu-satunya ekstrim relatif pada interval}

% Teorema 5.5.3, hlm. 178.
\begin{teorema}
Jika $f$ kontinu pada suatu interval $I$ dan hanya mempunyai satu ekstrim relatif di $c$, maka minimum relatif tersebut adalah minimum pada $I$, atau maksimum relatif tersebut adalah maksimum pada $I$.
\end{teorema}
\onslide<2->{\begin{contoh}
$f(x)=x+4/x$ pada $(0,\infty)$ mempunyai $f'=1-4/x^2$. Fungsi turun sebelum $2$ dan naik sesudah $2$. Jadi $f(2)=4$ adalah minimum absolut.
\end{contoh}
Kedua ujung menuju $+\infty$, sehingga maksimum tidak ada.}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Interval terbuka dan tak terbatas}

% Tabel 5.3 dan contoh 5.5.9--13, hlm. 177--182.
\begin{block}{Perilaku Ujung untuk Fungsi Kontinu pada $\mathbb R$}
\begin{itemize}
\item Kedua ujung menuju $+\infty$: ada minimum, tidak ada maksimum.
\item Kedua ujung menuju $-\infty$: ada maksimum, tidak ada minimum.
\item Ujung-ujung menuju tak hingga dengan tanda berlawanan: tidak ada maksimum maupun minimum.
\end{itemize}\end{block}
\onslide<2->{\begin{exampleblock}{Latihan}
Cari ekstrim $x+9/x$ pada $(0,\infty)$, lalu pada $[1,2]$. Jelaskan mengapa domain mengubah jawaban.
\end{exampleblock}}
\end{frame}

\section{Aplikasi Masalah Maksimum dan Minimum}

\begin{frame}

\framesubtitle{Dari cerita ke fungsi satu variabel}

% Hlm. 185--187: klasifikasi dan prosedur optimasi.
\begin{block}{Membangun Model Optimasi}\begin{enumerate}
\item Buat sketsa; tentukan besaran yang dioptimalkan.
\item Tulis fungsi tujuan: luas, volume, biaya, jarak, atau keuntungan.
\item Gunakan syarat masalah untuk menyisakan satu variabel.
\item Tentukan domain fisik, cari kandidat, dan uji maksimum/minimum.

\end{alertblock}}

\section{Teorema Rolle dan Teorema Nilai Rata-rata}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Rolle: kembali ke ketinggian yang sama}

% Teorema 5.7.1, hlm. 202. Bentuk f(a)=f(b) diperoleh dari
% bentuk f(a)=f(b)=0 di buku dengan mengurangi konstanta f(a).
\begin{teorema}[Rolle]
Jika $f$ kontinu pada $[a,b]$, dapat diturunkan pada $(a,b)$, dan $f(a)=f(b)$, maka ada $c\in(a,b)$ dengan $f'(c)=0$.
\end{teorema}
\onslide<2->{\begin{block}{Gagasan Bukti}
Jika grafik tidak konstan, salah satu maksimum atau minimum berada di interior. Di titik ekstrim yang dapat diturunkan, gradiennya nol.
Jika grafik konstan, setiap titik interior dapat dipilih sebagai $c$.
\end{block}}
\end{frame}

\framesubtitle{Teorema Nilai Rata-rata}

% Teorema 5.7.2, hlm. 204.
\begin{teorema}[Nilai Rata-rata]
Jika $f$ kontinu pada $[a,b]$ dan dapat diturunkan pada $(a,b)$, maka ada $c\in(a,b)$ sehingga
\[f'(c)=\frac{f(b)-f(a)}{b-a}.\]
\end{teorema}
\onslide<2->{\begin{block}{Arti geometris dan fisik}
Ada garis singgung yang sejajar garis potong kedua ujung. Untuk posisi sebagai fungsi waktu, ada saat ketika kecepatan sesaat sama dengan kecepatan rata-rata sepanjang interval waktu tersebut.
\end{block}}
\end{frame}

\begin{frame}
\frametitle{\insertsection}
\framesubtitle{Mengapa TNR mengikuti Rolle?}

% Bukti hlm. 204--205.
\begin{block}{Kurangi Grafik dengan Garis Potongnya}
Tuliskan $m=(f(b)-f(a))/(b-a)$ dan
\[g(x)=f(x)-[f(a)+m(x-a)].\]
Maka $g(a)=g(b)=0$, dan $g$ memenuhi syarat Rolle.
\end{block}
\onslide<2->{Ada $c\in(a,b)$ dengan $g'(c)=0$, sehingga $f'(c)-m=0$.
\begin{alertblock}{Yang dijamin adalah keberadaan}
Teorema tidak menjamin $c$ tunggal atau berada di titik tengah. Nilainya bergantung pada bentuk fungsi.
\end{alertblock}}
\end{frame}

\frametitle{\insertsection}
\framesubtitle{Akibat dan latihan penutup}

% Aplikasi latihan 5.7, hlm. 206--207.
\begin{block}{Turunan Sama, Selisih Konstan}
Jika $f'=g'$ pada suatu interval, terapkan TNR pada $h=f-g$. Karena $h'=0$, maka $h(b)=h(a)$ untuk setiap $a,b$: selisih $f-g$ konstan.
\end{block}
\begin{exampleblock}{Latihan}\begin{enumerate}
\item Cari semua $c$ yang dijamin Rolle untuk $\sin x$ pada $[0,2\pi]$.
\item Cari $c$ yang dijamin TNR untuk $x^3+2$ pada $[1,2]$.
\item Mobil bergerak lurus searah sejauh 12 km dalam 15 menit. Dengan asumsi posisi kontinu dan dapat diturunkan, jelaskan mengapa pernah terjadi kecepatan tepat 48 km/jam.
\end{enumerate}\end{exampleblock}
\end{frame}
