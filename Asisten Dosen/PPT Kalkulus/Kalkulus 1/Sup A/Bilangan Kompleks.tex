\section{Bilangan Kompleks}

% Bentuk aljabar, bidang Argand, dan operasi (hlm. 271--273).
\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Satuan imajiner dan bagian-bagian bilangan kompleks}
    Pada bilangan real, $x^2+1=0$ tidak mempunyai solusi. Dengan memperkenalkan satuan imajiner $i$, kita mendapatkan dua solusi, yaitu $i$ dan $-i$.
    \begin{definisi}[Bilangan kompleks]
        \[
            \C=\{a+bi\mid a,b\in\R\},\qquad i^2=-1.
        \]
        Untuk $z=a+bi$, $\Real(z)=a$ adalah \textbf{bagian real} dan
        $\Imag(z)=b$ adalah \textbf{bagian imajiner}.
    \end{definisi}
    \onslide<2->{%
        \begin{contoh}
            Jika $z=4-3i$, maka $\Real(z)=4$ dan $\Imag(z)=-3$.
            Bilangan $-2=-2+0i$ adalah real, sedangkan $3i=0+3i$ adalah imajiner murni.
        \end{contoh}
    }
    Setiap bilangan real dapat ditulis sebagai $a+0i$, sehingga $\R\subset\C$.
\end{frame}

            \end{tikzpicture}

                \[a+bi\ \longleftrightarrow\ (a,b).\]

            \end{block}

                \end{block}
            }
        \end{column}
    \end{columns}

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Empat operasi dalam bentuk $a+bi$}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{.6\textwidth}
            \begin{definisi}[Operasi pada bilangan kompleks]
                Untuk $z_1=a+bi$ dan $z_2=c+di$,
                \[
                    \begin{aligned}
                        z_1+z_2&=(a+c)+(b+d)i,\\
                        z_1-z_2&=(a-c)+(b-d)i,\\
                        z_1z_2&=(ac-bd)+(ad+bc)i,\\
                        \frac{z_1}{z_2}&=\frac{ac+bd}{c^2+d^2}
                        +\frac{bc-ad}{c^2+d^2}i,\quad z_2\ne0.
                    \end{aligned}
                \]
            \end{definisi}
        \end{column}
        \begin{column}{.37\textwidth}
            \onslide<2->{%
                \begin{block}{Asal suku $-bd$}
                    Hasil kali dua suku imajiner menjadi real karena
                    $(bi)(di)=bd\,i^2=-bd$.
                \end{block}
                Pada pembagian, $z_2\ne0$ memastikan $c^2+d^2>0$.
                Penyebut ini diperoleh dengan mengalikan $z_2$ dengan sekawannya.
            }
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Sekawan adalah pencerminan pada sumbu real}
    \begin{columns}[c]
        \begin{column}{.48\textwidth}
            \begin{definisi}[Sekawan atau konjugat]
                Untuk $z=a+bi$, sekawannya adalah $\bar z=a-bi$.
            \end{definisi}
            \onslide<2->{%
                \begin{contoh}
                    Jika $z=-2+3i$, maka $\bar z=-2-3i$.
                \end{contoh}
                Pencerminan dua kali menghasilkan $\overline{\bar z}=z$.
                Jika $b=0$, titik berada pada sumbu real dan $\bar z=z$.
            }
        \end{column}
        \begin{column}{.48\textwidth}
            \centering
            \begin{tikzpicture}[scale=.65]
                \draw[argand axis] (-3.8,0)--(2.5,0) node[right] {$\Real$};
                \draw[argand axis] (0,-3.45)--(0,3.45) node[above] {$\Imag$};
                \draw[argand guide] (-2,-3)--(-2,3);
                \draw[argand vector] (0,0)--(-2,3);
                \node[argand point,label=above left:{$z$}] at (-2,3) {};
                \onslide<2->{%
                    \draw[argand vector,magenta] (0,0)--(-2,-3);
                    \node[argand point,fill=magenta,label=below left:{$\bar z$}] at (-2,-3) {};
                }
                \node[below right] at (0,0) {$0$};
            \end{tikzpicture}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Mengambil bagian real dan imajiner dengan sekawan}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{.48\textwidth}
            \begin{block}{Untuk $z=a+bi$}
                \[
                    \begin{aligned}
                        z+\bar z&=2a, &
                        \Real(z)&=\frac{z+\bar z}{2},\\
                        z-\bar z&=2bi, &
                        \Imag(z)&=\frac{z-\bar z}{2i}.
                    \end{aligned}
                \]
            \end{block}
            \onslide<2->{Penjumlahan menghapus suku imajiner, sedangkan pengurangan menghapus suku real.}
        \end{column}
        \begin{column}{.48\textwidth}
            \begin{block}{Hasil kali dengan sekawan}
                \[
                    z\bar z=(a+bi)(a-bi)=a^2+b^2.
                \]
                Nilainya real dan taknegatif.
            \end{block}
            \onslide<2->{%
                \begin{contoh}
                    Untuk $z=-2+3i$, berlaku $z+\bar z=-4$, $z-\bar z=6i$,
                    dan $z\bar z=13$.
                \end{contoh}
            }
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Sekawan pada hasil operasi}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{.46\textwidth}
            \begin{block}{Sifat sekawan}
                \[
                    \begin{aligned}
                        \overline{z_1+z_2}&=\bar z_1+\bar z_2,\\
                        \overline{z_1-z_2}&=\bar z_1-\bar z_2,\\
                        \overline{z_1z_2}&=\bar z_1\bar z_2,\\[2pt]
                        \overline{\left(\frac{z_1}{z_2}\right)}
                        &=\frac{\bar z_1}{\bar z_2},\quad z_2\ne0.
                    \end{aligned}
                \]
            \end{block}
        \end{column}
        \begin{column}{.5\textwidth}
            % Adaptasi soal latihan hlm. 276.
            \begin{contoh}
                Mengapa $\displaystyle\frac{2+3i}{1-i}+\frac{2-3i}{1+i}$ real?
            \end{contoh}
            \onslide<2->{%
                Kedua suku saling bersekawan. Jika suku pertama $u$, jumlahnya
                $u+\bar u=2\Real(u)\in\R$.
            }
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}

                    -- (2.65,1.9) -- (0,1.9) node[left] {$b$};
                \draw[argand vector] (0,0)--(2.65,1.9) node[midway,above left] {$r$};
                \node[argand point,label=above:{$z=a+bi$}] at (2.65,1.9) {};

            \end{tikzpicture}
        \end{column}
        \begin{column}{.49\textwidth}
            \begin{definisi}[Bentuk kutub]
                Untuk $z=a+bi\ne0$, ambil $r=|z|$. Jika $\theta$ adalah sudut berarah dari sumbu real positif ke vektor $z$, maka
                \[
                    z=r(\cos\theta+i\sin\theta).
                \]
                Sudut $\theta$ disebut \textbf{argumen} dari $z$.
            \end{definisi}
            \onslide<2->{%
                Proyeksi memberi $a=r\cos\theta$ dan $b=r\sin\theta$.
                Notasi lain adalah $z=r\cis\theta=re^{i\theta}$, dengan $\theta$ dalam radian untuk eksponen.
            }
        \end{column}
    \end{columns}
    \par\smallskip
    \onslide<2->{Untuk $z=0$, modulusnya nol dan argumennya tidak terdefinisi.}
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Argumen utama ditentukan oleh kuadran}
    Untuk $z=a+bi\ne0$, pilih argumen utama $\Arg(z)$ dalam interval $(0,2\pi]$.
    Semua argumen $z$ berbentuk $\theta=\Arg(z)+2k\pi$ dengan $k\in\mathbb Z$.
    \par\smallskip

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Aturan untuk kali, bagi, dan pangkat}
    \begin{teorema}[De Moivre]
        Misalkan $z_j=r_j\cis\theta_j$ untuk $j=1,\ldots,m$ ($m\ge2$) dan
        $z=r\cis\theta$, dengan semua modulus positif.
        \[
            \begin{aligned}
                z_1z_2&=r_1r_2\cis(\theta_1+\theta_2), &
                \frac{z_1}{z_2}&=\frac{r_1}{r_2}\cis(\theta_1-\theta_2),\\
                z_1\cdots z_m&=(r_1\cdots r_m)\cis(\theta_1+\cdots+\theta_m),&
                z^n&=r^n\cis(n\theta),\quad n\in\mathbb Z.
            \end{aligned}
        \]
    \end{teorema}
    \onslide<2->{%
        \begin{block}{Arti perkalian di bidang kompleks}
            Mengalikan dengan $r_2\cis\theta_2$ mengubah jarak dari titik asal dengan faktor $r_2$ dan memutar arah sebesar $\theta_2$.
            Pembagian membalik kedua perubahan itu.
        \end{block}
    }
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Identitas trigonometri menjelaskan aturan sudut}
    Mulai dengan bentuk bermodulus $1$. Distribusikan suku, lalu pakai $i^2=-1$.
    \[
        \begin{aligned}
            (\cis\alpha)(\cis\beta)
                &= (\cos\alpha\cos\beta-\sin\alpha\sin\beta)\\
                &\quad+i(\sin\alpha\cos\beta+\cos\alpha\sin\beta)\\
                &=\cos(\alpha+\beta)+i\sin(\alpha+\beta).
        \end{aligned}
    \]
    \onslide<2->{%
        \begin{columns}[T,onlytextwidth]
            \begin{column}{.48\textwidth}
                \begin{block}{Membagi}
                    $\cis\beta\cdot\cis(-\beta)=1$.
                    Jadi membagi dengan $r_2\cis\beta$ sama dengan mengalikan oleh $(1/r_2)\cis(-\beta)$.
                \end{block}
            \end{column}
            \begin{column}{.48\textwidth}
                \begin{block}{Memangkatkan}
                    Untuk $n>0$, kalikan berulang. Untuk $n<0$, gunakan $z^n=1/z^{-n}$.
                    Untuk $n=0$, rumus memberi $1$.
                \end{block}
            \end{column}
        \end{columns}
    }
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Semua akar pangkat $n$ dari satu bilangan}
    \begin{definisi}[Akar pangkat $n$]
        Untuk bilangan bulat $n\ge1$, $w$ disebut akar pangkat $n$ dari $z$ jika $w^n=z$.
        Jika $z=r\cis\theta$ dengan $r>0$, akar-akarnya adalah
        \[
            w_k=r^{1/n}\cis\!\left(\frac{\theta+2k\pi}{n}\right),
            \qquad k=0,1,\ldots,n-1.
        \]
    \end{definisi}
    \onslide<2->{%
        \begin{block}{Letak akar di bidang Argand}
            Semua akar berjarak $r^{1/n}$ dari titik asal. Dari satu akar ke akar berikutnya,
            sudut bertambah $2\pi/n$. Untuk $n\ge3$, akar merupakan titik sudut segi-$n$ beraturan.
        \end{block}
    }
    Untuk $z=0$, satu-satunya akar pangkat $n$ adalah $w=0$.
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Mengapa jumlahnya tepat $n$?}
    Jika $w=\rho\cis\varphi$ dan $z=r\cis\theta\ne0$, Teorema De Moivre mengubah $w^n=z$ menjadi
    \[
        \rho^n=r,\qquad n\varphi=\theta+2k\pi,\quad k\in\mathbb Z.
    \]
    \onslide<2->{%
        Syarat pertama menetapkan $\rho=r^{1/n}$. Syarat kedua memberi
        $\varphi=(\theta+2k\pi)/n$, karena sudut yang berbeda $2\pi$ menunjuk arah yang sama.
        \begin{block}{Satu putaran memuat $n$ akar berbeda}
            Kenaikan $k$ sebesar $1$ menggeser sudut sebesar $2\pi/n$.
            Setelah $n$ langkah, sudut bertambah $2\pi$ dan titiknya kembali ke akar pertama.
        \end{block}
    }
    \onslide<3->{Pemeriksaan dengan pemangkatan memberi $w_k^n=r\cis(\theta+2k\pi)=z$.}
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Akar kuadrat dari komponen real dan imajiner}
    Jika $z=r\cis\theta\ne0$, dua akarnya adalah $\pm\sqrt r\,\cis(\theta/2)$.
    Untuk mencari langsung dari $z=a+bi$, tulis $w=p+qi$.
    \begin{columns}[c,onlytextwidth]
        \begin{column}{.46\textwidth}
            \begin{block}{Persamaan dari $w^2=z$}
                \[
                    p^2-q^2=a,\qquad 2pq=b.
                \]
                Dari modulus diperoleh $p^2+q^2=|w|^2=r$.
                Gabungkan persamaan jumlah dan selisih untuk mendapatkan $p^2$ dan $q^2$.
            \end{block}
        \end{column}
        \begin{column}{.5\textwidth}
            \onslide<2->{%
                \begin{block}{Memilih tanda saat $b\ne0$}
                    Ambil akar positif untuk $p$,
                    \[
                        p=\sqrt{\frac{r+a}{2}},\qquad
                        |q|=\sqrt{\frac{r-a}{2}}.
                    \]
                    Gunakan $q=b/(2p)$, sehingga tanda $q$ mengikuti tanda $b$.
                    Kedua akar adalah $w$ dan $-w$.
                \end{block}
            }
        \end{column}
    \end{columns}
    \par\smallskip
    \onslide<3->{Untuk $b=0$, akarnya $\pm\sqrt a$ jika $a\ge0$ dan $\pm i\sqrt{-a}$ jika $a<0$.}
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \framesubtitle{Rumus kuadrat berlaku untuk koefisien kompleks}
    Misalkan $az^2+bz+c=0$ dengan $a,b,c\in\C$ dan $a\ne0$.
    Lengkapi kuadrat untuk memperoleh
    \[
        \left(z+\frac{b}{2a}\right)^2=\frac{b^2-4ac}{4a^2}
        \quad\Longleftrightarrow\quad (2az+b)^2=\Delta,
        \qquad \Delta=b^2-4ac.
    \]
    \onslide<2->{%
        \begin{block}{Rumus kuadrat}
            Ambil salah satu akar kuadrat $s$ dari $\Delta$. Kedua solusi diberikan oleh
            \[
                z=\frac{-b\pm s}{2a},\qquad s^2=\Delta.
            \]
            Jika $\Delta=0$, kedua tanda memberi akar kembar yang sama.
        \end{block}
    }
    \onslide<3->{Untuk koefisien real, mengambil sekawan pada persamaan menunjukkan bahwa akar nonreal muncul berpasangan sekawan.}
\end{frame}
