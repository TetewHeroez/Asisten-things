\section{Notasi Limit}
\begin{frame}
    \frametitle{\insertsection}
    \begin{definisi}
        Notasi limit yang biasanya dibaca ``limit $f(x)$ saat $x$ \textbf{mendekati} $a$ adalah $L$'' dituliskan sebagai
        \[\lim_{x\to a} f(x)=L,\]
        Artinya jika kita mengambil nilai $x$ yang sangat dekat dengan $a$, maka $f(x)$ akan sangat dekat dengan $L$.
    \end{definisi}
    \onslide<2->{
            \setbeamercolor{item}{fg=PastelGreen!50!HIMAmuda}
            \begin{alertblock}{Catatan}
                \begin{itemize}
                    \item Kata ``mendekati'' jangan disamakan dengan ``menuju''.
                    \item Nilai $f(a)$ tidak harus sama dengan $L$ atau bahkan $f(a)$ tidak terdefinisi.
                    \item Nilai $f(x)$ untuk $x=a$ tidak mempengaruhi nilai limit.
                \end{itemize}
            \end{alertblock}}
\end{frame}

\section{Perhitungan Limit}
\begin{frame}
    \frametitle{\insertsection}
    \begin{teorema}
        Misalkan $f$ dan $g$ adalah dua fungsi dengan $\displaystyle\lim_{x\to a} f(x)=L$ dan $\displaystyle\lim_{x\to a} g(x)=M$, maka
        \begin{enumerate}
            \item $\displaystyle\lim_{x\to a} [f(x)+g(x)]=L+M$
            \item $\displaystyle\lim_{x\to a} [f(x)-g(x)]=L-M$
            \item $\displaystyle\lim_{x\to a} [f(x)\cdot g(x)]=L\cdot M$
            \item Jika $M\ne 0$, maka $\displaystyle\lim_{x\to a} \frac{f(x)}{g(x)}=\frac{L}{M}$
            \item $\displaystyle\lim_{x\to a} \sqrt[n]{f(x)}=\sqrt[n]{L}$, dengan $n$ bilangan bulat positif
        \end{enumerate}
    \end{teorema}
\end{frame}
\begin{frame}
    \frametitle{\insertsection}
    \begin{contoh}
        \begin{itemize}
            \item $\displaystyle\lim_{x\to 2} x=2$
            \item $\displaystyle\lim_{x\to \frac{1}{2}} \frac{6x^2-x-1}{2x-1}=\frac{5}{2}$
            \item $\displaystyle\lim_{x\to 3} \frac{\sqrt{12-x}-3}{x-3}=-\frac{1}{6}$
            \item $\displaystyle\lim_{x\to 0} \frac{\sin x}{x}=1$
        \end{itemize}
    \end{contoh}
\end{frame}
\begin{frame}
    \frametitle{\insertsection}
    \begin{exampleblock}{Latihan}
        Hitunglah nilai limit berikut:
        \begin{enumerate}
            \item $\displaystyle\lim_{x\to 4} \frac{\sqrt{x}-2}{x-4}$
            \item $\displaystyle\lim_{x\to 1} \frac{x^2-1}{x^2-x}$
            \item $\displaystyle\lim_{x\to -3} \frac{x^2+x-6}{x+3}$
        \end{enumerate}
    \end{exampleblock}
\end{frame}

\section{Limit di Tak-Hingga}
\begin{frame}
    \frametitle{\insertsection}
    \begin{definisi}[Limit Menuju Tak-Hingga]
        Kita menuliskan \[\lim_{x\to \infty} f(x) = L\]
        jika nilai $f(x)$ dapat dibuat sedekat mungkin ke $L$ dengan mengambil $x$ yang cukup besar. Garis $y=L$ disebut \textbf{Asimtot Datar} dari kurva $y=f(x)$.
    \end{definisi}
    \onslide<2->{
        \begin{teorema}
            Jika $r > 0$ adalah bilangan rasional, maka
            \[\lim_{x\to \infty} \frac{1}{x^r} = 0 \quad \text{dan} \quad \lim_{x\to -\infty} \frac{1}{x^r} = 0\]
        \end{teorema}
    }
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \begin{contoh}
        Tentukan limit berikut:
        \begin{enumerate}
            \item $\displaystyle\lim_{x\to \infty} \frac{3x^2-x-2}{5x^2+4x+1}$
            \item $\displaystyle\lim_{x\to \infty} \frac{\sqrt{2x^2+1}}{3x-5}$
        \end{enumerate}
    \end{contoh}
    \onslide<2->{
        \textbf{Jawab:}
        Bagi pembilang dan penyebut dengan pangkat tertinggi dari $x$ di penyebut.
        \begin{enumerate}
            \item $\displaystyle\lim_{x\to \infty} \frac{3x^2-x-2}{5x^2+4x+1} = \lim_{x\to \infty} \frac{3 - \frac{1}{x} - \frac{2}{x^2}}{5 + \frac{4}{x} + \frac{1}{x^2}} = \frac{3-0-0}{5+0+0} = \frac{3}{5}$
            \item $\displaystyle\lim_{x\to \infty} \frac{\sqrt{2x^2+1}}{3x-5} = \lim_{x\to \infty} \frac{\sqrt{2+1/x^2}}{3-5/x} = \frac{\sqrt{2}}{3}$
        \end{enumerate}
    }
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \begin{definisi}[Limit Bernilai Tak-Hingga]
        Kita menuliskan \[\lim_{x\to a} f(x) = \infty\]
        jika nilai $f(x)$ dapat dibuat sebesar mungkin (positif) dengan mengambil $x$ yang cukup dekat ke $a$ namun $x \ne a$. Garis $x=a$ disebut \textbf{Asimtot Tegak} dari kurva $y=f(x)$.
    \end{definisi}
    \onslide<2->{
        \begin{contoh}
            Tentukan $\displaystyle\lim_{x\to 3^+} \frac{2x}{x-3}$ dan $\displaystyle\lim_{x\to 3^-} \frac{2x}{x-3}$.\\
            Saat $x \to 3^+$, $x-3 \to 0$ (positif). Maka $\displaystyle\lim_{x\to 3^+} \frac{2x}{x-3} = \infty$.\\
            Saat $x \to 3^-$, $x-3 \to 0$ (negatif). Maka $\displaystyle\lim_{x\to 3^-} \frac{2x}{x-3} = -\infty$.
        \end{contoh}
    }
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \begin{exampleblock}{Latihan}
        \begin{enumerate}
            \item Tentukan asimtot datar dan asimtot tegak dari fungsi $\displaystyle f(x) = \frac{2x^2+1}{x^2-1}$
            \item Hitunglah $\displaystyle\lim_{x\to \infty} \left(\sqrt{x^2+x} - x\right)$
            \item Hitunglah $\displaystyle\lim_{x\to 2^-} \frac{x^2-2x}{x^2-4x+4}$
        \end{enumerate}
    \end{exampleblock}
\end{frame}

\section{Kekontinuan}
\begin{frame}
    \frametitle{\insertsection}
    \begin{definisi}[Fungsi Kontinu di Suatu Titik]
        Suatu fungsi $f$ dikatakan kontinu di titik $x = a$ jika memenuhi ketiga syarat berikut:
        \begin{enumerate}
            \item $f(a)$ terdefinisi (artinya $a \in \dom(f)$)
            \item $\displaystyle\lim_{x\to a} f(x)$ ada
            \item $\displaystyle\lim_{x\to a} f(x) = f(a)$
        \end{enumerate}
        Jika salah satu atau lebih syarat di atas tidak terpenuhi, maka $f$ dikatakan \textbf{diskontinu} di titik $a$.
    \end{definisi}
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    Ada tiga jenis diskontinuitas yang sering dijumpai:
    \begin{itemize}
        \item \textbf{Diskontinuitas yang dapat dihapus (Removable):} $\displaystyle\lim_{x\to a} f(x)$ ada, tetapi nilainya tidak sama dengan $f(a)$ atau $f(a)$ tidak terdefinisi.
        \item \textbf{Diskontinuitas loncat (Jump):} Limit kiri dan limit kanan ada, tetapi nilainya berbeda.
        \item \textbf{Diskontinuitas tak-hingga (Infinite):} Limit kiri atau limit kanan (atau keduanya) bernilai tak-hingga.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \begin{contoh}
        Tentukan apakah fungsi berikut kontinu di $x=2$:
        \[f(x) = \begin{cases}
            \frac{x^2-4}{x-2}, & x \ne 2 \\
            4, & x = 2
        \end{cases}\]
        \onslide<2->{
        \textbf{Jawab:}
        \begin{enumerate}
            \item $f(2) = 4$ (terdefinisi)
            \item $\displaystyle\lim_{x\to 2} f(x) = \lim_{x\to 2} \frac{x^2-4}{x-2} = \lim_{x\to 2} (x+2) = 4$ (limit ada)
            \item $\displaystyle\lim_{x\to 2} f(x) = f(2) = 4$
        \end{enumerate}
        Karena ketiga syarat terpenuhi, maka $f$ kontinu di $x=2$.
        }
    \end{contoh}
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \begin{teorema}[Teorema Nilai Antara]
        Misalkan $f$ adalah fungsi yang kontinu pada interval tertutup $[a,b]$, dan misalkan $N$ adalah bilangan di antara $f(a)$ dan $f(b)$ (dimana $f(a) \ne f(b)$). Maka terdapat suatu bilangan $c$ di dalam interval terbuka $(a,b)$ sedemikian sehingga $f(c) = N$.
    \end{teorema}
    \onslide<2->{
        \begin{alertblock}{Penerapan Penting}
            Jika fungsi $f$ kontinu di interval $[a,b]$ dan nilai $f(a)$ dengan $f(b)$ berlawanan tanda ($f(a)f(b) < 0$), maka persamaan $f(x)=0$ memiliki \textbf{setidaknya satu akar} pada interval $(a,b)$.
        \end{alertblock}
    }
\end{frame}

\begin{frame}
    \frametitle{\insertsection}
    \begin{exampleblock}{Latihan}
        \begin{enumerate}
            \item Tentukan nilai $k$ agar fungsi berikut kontinu di $x=3$:
            \[f(x) = \begin{cases}
                \frac{x^2-9}{x-3}, & x \ne 3 \\
                k, & x = 3
            \end{cases}\]
            \item Tentukan titik-titik di mana fungsi $\displaystyle f(x) = \frac{x+1}{x^2-4x+3}$ tidak kontinu, dan tentukan jenis diskontinuitasnya.
            \item Tunjukkan bahwa persamaan $x^3 - 4x + 1 = 0$ memiliki sedikitnya satu solusi pada interval $[1,2]$.
        \end{enumerate}
    \end{exampleblock}
\end{frame}
